```python
def recived_total(*quantities):
    a = 0
    for i in quantities:
        a = a + i
    return a

batches = [2, 4, 1]
print(recived_total(3, 0, 5))  # 8
print(recived_total(*batches)) # 7
print(recived_total())        # 0
```
