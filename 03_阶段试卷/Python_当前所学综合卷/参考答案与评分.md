## C01 · 10分

以下用函数包装方便换数据验证，也可直接写等价脚本。这里函数里的print就是C题要求的三行输出。

```python
def arrival_report(commands, initial_stock, box_size):
    stock = initial_stock
    index = 0
    while index < len(commands):
        command = commands[index]
        index += 1
        if command == "stop":
            break
        quantity = int(command)
        if quantity <= 0:
            continue
        stock += quantity
    print(f"总库存：{stock}件")
    print(f"整箱：{stock // box_size}箱")
    print(f"剩余：{stock % box_size}件")

arrival_report(["3", "0", "-2", "5", "stop", "100"], 2, 4)
```

| stop用break停止，后续不处理 | 2 |
| 数字转int，用continue跳过<=0 | 2 |
| 从实际初始库存累加正数，不修改commands | 1 |
| 用实际box_size计算整除和余数，输出格式正确 | 2 |
| 空列表、首条stop、无stop均正确结束并保留应有库存 | 1 |

def session_summary(sessions, min_minutes=30):
    selected_minutes = []
    for topic, minutes in sessions:
        if minutes >= min_minutes:
            selected_minutes.append(minutes)
    total = sum_minutes(*selected_minutes)
    count = len(selected_minutes)
    average = total / count if count else None
    return count, total, average

## E · 20分

```python
def parse_signins(raw):
    records = []
    for part in raw.split(";"):
        part = part.strip()
        if part:
            name, topic = part.split(",")
            records.append((name.strip().lower(), topic.strip().lower()))
    return records

def group_topics(records):
    grouped = {}
    for name, topic in records:
        if name not in grouped:
            grouped[name] = set()
        grouped[name].add(topic)
    return grouped

def rank_learners(grouped, min_topics=1):
    ranked = []
    for name in sorted(grouped):
        count = len(grouped[name])
        if count >= min_topics:
            ranked.append((name, count))
    return sorted(ranked, key=lambda row: row[1], reverse=True)

def learning_report(raw, min_topics=1):
    records = parse_signins(raw)
    grouped = group_topics(records)
    ranked = rank_learners(grouped, min_topics=min_topics)
    topics = set()
    for name, count in ranked:
        for topic in grouped[name]:
            topics.add(topic)
    return {
        "record_count": len(records),
        "learner_count": len(ranked),
        "topic_names": sorted(topics),
        "ranked": ranked,
    }

raw = " Ada ,Python;bo,SQL;ADA,python;;cy,Python; Bo ,Linux;ada,sql; \n"
print(learning_report(raw))
print(learning_report(raw, min_topics=2))
print(learning_report(raw, min_topics=3))
print(learning_report(""))
```
